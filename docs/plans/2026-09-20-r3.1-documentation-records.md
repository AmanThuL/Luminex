# R3.1 Documentation Records Implementation Plan

**Status**: Accepted

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:subagent-driven-development. Steps use
> checkbox (`- [ ]`) syntax. Set the status to `In progress` when Wave 0 starts.

**Goal:** Regroup `docs/` into series folders, delete the closed plans, remove private-note
mentions, teach the checker the new layout, and copyedit the living documents.

**Architecture:** A resolver script (below) does every move and reference rewrite from one table.
Hand edits happen in parallel waves over disjoint file sets. The main thread owns git.

**Tech stack:** Python 3 (`unittest`), git, Markdown. No build is needed.

**Spec:** [`docs/milestones/r3.1.md`](../milestones/r3.1.md). Read it first; its rules bind every task.

## Global constraints

- Branch `r3.1-documentation-records`, in place. **Subagents never run `git add`, `commit`,
  `stash`, `checkout`, `restore` or `mv`.** They edit only the files their task lists, then report.
  The main thread reviews and commits. This is what makes the parallel waves safe.
- Gate at every commit: `python3 Tools/check_project_policy.py --commits main..HEAD` and
  `python3 -m unittest discover -s Tools/tests`, both green.
- Commit subjects: `<scope>: <lowercase imperative>`, at most 72 characters, scope `docs` or `tool`.
  No AI trailer. Messages never contain the words the checker's `PROCESS_PATTERNS` rejects.
- Frozen designs, research, postmortems and accepted ADRs change in path references only.
- Tracked files never mention the owner's private notes, including this plan and commit messages.
- Line budgets: `AGENTS.md` and `README.md` 250; conventions, guides, architecture, milestones
  (except `-design.md`), plans and roadmap files 300.
- Models: Opus main thread. Sonnet for T1, T3, T4, T5, T7 and prose tasks H2 to H4 and H6 to H10.
  Opus for T6, H1 and H5. Opus reviews Sonnet work; the main thread reviews Opus work.

## Move script

Save as `move_docs.py` in the session scratch directory. Proven on a clone: 68 renames, 340
insertions against 340 deletions, no broken link.

```python
#!/usr/bin/env python3
"""Move one group of documents and rewrite every reference. Run from the repository root."""
import os, re, subprocess, sys
from pathlib import Path

M, S, P = "docs/milestones/", "docs/specs/", "docs/postmortems/"
DESIGNS = {
    "m5": "2026-08-11-m5-execution-substrate", "m5.1": "2026-08-12-m5.1-rhi-execution-model",
    "m5.2": "2026-08-12-m5.2-rhi-frame-data", "m5.3": "2026-08-13-m5.3-editor-workspace-selection",
    "m5.4": "2026-09-04-m5.4-render-graph-node-view", "m5.5": "2026-09-05-m5.5-graph-legibility",
    "m5.6": "2026-09-06-m5.6-gpu-work-submission", "m6.1": "2026-09-07-m6.1-temporal-state-motion",
    "m6.2": "2026-09-07-m6.2-native-taa-exposure", "m6.3": "2026-09-09-m6.3-taau-dynamic-resolution",
    "m6.4": "2026-09-12-m6.4-metalfx-temporal-adapter", "m6.5": "2026-09-12-m6.5-display-boundary-edr",
    "m7.1": "2026-09-15-m7.1-gpu-scene-foundation", "r1.4": "2026-09-13-r1.4-renderer-seams",
    "r1.5": "2026-09-13-r1.5-consolidation", "r2.3": "2026-09-19-r2.3-mechanical-rename",
    "r2.4": "2026-09-20-r2.4-extract-mount-wire", "ux1": "2026-09-14-ux1-editor-experience",
}
RECORDS = {
    "m1-m4": "m1 m2 m3 m3.1 m4 m4.1",
    "m5": "m5 m5.1 m5.2 m5.3 m5.4 m5.5 m5.6",
    "m6": "m6.1 m6.2 m6.3 m6.4 m6.5 interface-gate-b",
    "m7": "m7.1 m7.1-validation-history m7.2 m7.2-editor-transport m7.2-performance-window "
          "m7.2-validation m7.2-visibility-followup m7.3 m7.3-validation m7.4 m7.4-editor-validation "
          "m7.4-validation m7.5 m7.5-followup m7.5-validation",
    "r": "r1.1 r1.2 r1.3 r1.4 r1.5 r2 r2.3-validation r2.4-validation r3 r3.1 r4",
    "ux": "ux1 ux2",
}
SERIES = {"m5": "m5", "m6": "m6", "m7": "m7", "r1": "r", "r2": "r", "ux": "ux"}
EXTRA = {
    "m1-m4": {P + "2026-08-09-low-shadow-contrast.md": M + "m1-m4/2026-08-09-low-shadow-contrast.md"},
    "m6": {P + "2026-09-08-blocky-bloom-and-reflections.md":
           M + "m6/2026-09-08-blocky-bloom-and-reflections.md"},
    "founding": {S + "2026-08-07-luminex-upgrade-design.md": "docs/decisions/0000-founding-design.md"},
}
LINK = re.compile(r"(\]\(<?)([^)\s>]+)")
BARE = re.compile(r"(?<![\w(</])()((?:\.\./)+[\w.\-/]+\.md)")


def group(name):
    moves = dict(EXTRA.get(name, {}))
    for record in RECORDS.get(name, "").split():
        moves[f"{M}{record}.md"] = f"{M}{name}/{record}.md"
    for ident, stem in DESIGNS.items():
        if SERIES[ident[:2]] == name:
            moves[f"{S}{stem}-design.md"] = f"{M}{name}/{ident}-design.md"
    return moves


def rewrite(path, moves, back):
    file = Path(path)
    if not file.is_file():
        return
    try:
        text = file.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        return
    old_dir, new_dir = os.path.dirname(back.get(path, path)), os.path.dirname(path)

    def link(match):
        target, _, anchor = match.group(2).partition("#")
        if not target or re.match(r"[a-z]+:", target):
            return match.group(0)
        resolved = os.path.normpath(os.path.join(old_dir, target))
        if resolved not in moves and old_dir == new_dir:
            return match.group(0)
        fresh = os.path.relpath(moves.get(resolved, resolved), new_dir or ".")
        return match.group(1) + fresh + ("#" + anchor if anchor else "")

    fresh = BARE.sub(link, LINK.sub(link, text)) if path.endswith(".md") else text
    for old in sorted(moves, key=len, reverse=True):
        fresh = fresh.replace(old, moves[old])
    if fresh != text:
        file.write_text(fresh, encoding="utf-8")


moves = group(sys.argv[1]) or sys.exit(f"unknown group {sys.argv[1]}")
for old, new in moves.items():
    os.makedirs(os.path.dirname(new), exist_ok=True)
    subprocess.run(["git", "mv", old, new], check=True)
back = {new: old for old, new in moves.items()}
listing = subprocess.run(["git", "ls-files", "-z"], check=True, capture_output=True, text=True)
for path in listing.stdout.split("\0"):
    if path and not path.startswith(("RojoRHI/", "ThirdParty/")):
        rewrite(path, moves, back)
print(f"{sys.argv[1]}: moved {len(moves)} files")
```
